import re
import sys

targets = [
    ('artifacts/api-server/src/routes/doctrine.ts',       [(1210, 'string'),              (1282, 'string')]),
    ('artifacts/api-server/src/routes/gmail-auth.ts',     [(204,  'string'),              (228,  'string')]),
    ('artifacts/api-server/src/routes/email-inspector.ts',[(370,  'string | undefined')]),
]

# Match a bare req.query.<identifier> NOT already followed by ' as <word>'
# and NOT preceded by 'String(' — both guards make this idempotent.
pattern = re.compile(r'(?<!String\()\b(req\.query\.[a-zA-Z_][a-zA-Z0-9_]*)\b(?!\s+as\s+\w)')

errors = []
for fp, line_specs in targets:
    print(f"\n--- {fp} ---")
    try:
        with open(fp) as f:
            lines = f.readlines()
    except FileNotFoundError:
        print(f"  ERROR: file not found")
        errors.append(fp)
        continue

    changed = False
    for ln, target_type in line_specs:
        idx = ln - 1
        if idx >= len(lines):
            print(f"  WARN line {ln} beyond file end ({len(lines)} lines)")
            continue
        original = lines[idx]
        cast = ' as string' if target_type == 'string' else ' as string | undefined'
        new = pattern.sub(lambda m: f"({m.group(1)}{cast})", original)
        if new != original:
            print(f"  Line {ln}:")
            print(f"    OLD: {original.rstrip()}")
            print(f"    NEW: {new.rstrip()}")
            lines[idx] = new
            changed = True
        else:
            print(f"  NOOP line {ln} — pattern did not match (already cast, or different shape)")
            print(f"    Content: {original.rstrip()}")
    if changed:
        with open(fp, 'w') as f:
            f.writelines(lines)
        print(f"  Wrote {fp}")
PYEOF
